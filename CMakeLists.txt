# Nothing here throws or catches. Qt itself does not throw, it reports errors
# through return values, and an exception leaving a slot is undefined
# behaviour, so one could only ever end the process. The landing pads and
# exception tables for it were 201 KB of the binary. Unwind tables stay, so
# backtraces and crash reports still walk these frames, and Qt's headers
# follow the flag through QT_NO_EXCEPTIONS.
# https://doc.qt.io/qt-6/exceptionsafety.html
target_compile_options(noir PRIVATE -fno-exceptions)
target_link_options(noir PRIVATE "$<$<CONFIG:Release>:-Oz;-Wl,--gc-sections;-Wl,-z,pack-relative-relocs>")
set_target_properties(noir PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
# Every component is precompiled, including the first-use-only picker, because
# the module below keeps no QML source in the binary for the engine to fall
# back to. Its bytecode costs less than the source it replaces.
set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
set_target_properties(noir PROPERTIES QT_QMLCACHEGEN_ARGUMENTS "--only-bytecode")
# DISCARD_QML_CONTENTS drops the QML source from the resource. The engine runs
# the compiled units, so the text was only ever a fallback nothing reaches.
qt_add_qml_module(noir URI NoirUi VERSION 1.0 NO_PLUGIN NO_RESOURCE_TARGET_PATH DISCARD_QML_CONTENTS RESOURCE_PREFIX "/"
